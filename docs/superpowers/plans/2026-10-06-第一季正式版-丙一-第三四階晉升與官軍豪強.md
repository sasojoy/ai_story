class PromotionCast(_Strict):
    """一段奇遇的一個版本（晉升奇遇文件第一節「人物不在」與「時局」）：照順序第一個成立的演。
    figure 寫了：那位人物要在場（active）；at 也寫了就要在 at，沒寫 at 就是他此刻的所在。
    before_event：那件時刻表大事還沒結算才成立（何進的索賄在盧植下獄之前、之後各一版）。
    flags_none：這些世界旗標都不在才成立。after：上一段演的是這一則才成立（董卓版的第二段接董卓版的第一段）。"""

    event: str
    figure: str | None = None
    at: str | None = None
    before_event: str | None = None
    flags_none: list[str] = Field(default_factory=list)
    after: str | None = None
    summons_text: str  # 召見的話（{據點}＝地點名）


class PromotionLeg(_Strict):
    """晉升奇遇的一段（一個地點的戲）。location 是 cast 沒寫 at、也沒有 figure 時的地點（可寫 "nearest_base"）。"""

    location: str | None = None
    casts: list[PromotionCast]


class EventMod(_Strict):
    """一般伏筆（伏筆文件 4.4）：event 那件大事 side 那一方的成功率 +amount；那件大事還沒結算才算，全服合計夾在 ±0.20
    （timetable.add_mod）。"""

    event: str
    side: Literal["guan", "huang"]
    amount: float


class PatronLine(_Strict):
    """豪強升第 4 階時，看靠山多一句話（晉升奇遇文件 4.3）。"""

    text: str
    affinity: dict[str, int] = Field(default_factory=dict)
```

def threshold(content: Content, rank: int) -> int:
    cfg = content.config
    return {2: cfg.rank2_contrib, 3: cfg.rank3_contrib, 4: cfg.rank4_contrib}[rank]


def _cast_ok(state: GameState, content: Content, cast: PromotionCast, prev: str | None) -> bool:
    w = state.world
    if cast.after is not None and cast.after != prev:
        return False
    if cast.before_event is not None and cast.before_event in w.timeline:
        return False
    if any(flag in w.flags for flag in cast.flags_none):
        return False
    if cast.figure is not None:
        now = figures.state_of(state, content, cast.figure)
        if now.status != "active" or not now.location:
            return False
        if cast.at is not None and now.location != cast.at:
            return False
    return True


def current_cast(
    state: GameState, content: Content, promo: PromotionDef, leg: int, prev: str | None,
) -> tuple[PromotionCast, str] | None:
    """這一段此刻該演的版本與地點：照順序第一個成立的（晉升奇遇文件第一節）。地點：cast.at；沒寫 at 但有 figure 是
    那位此刻的所在；都沒寫是這一段的 location（nearest_base 照 T5 挑最近的投靠點）。沒有成立的是 None。"""
    if leg >= len(promo.legs):
        return None
    part = promo.legs[leg]
    for cast in part.casts:
        if not _cast_ok(state, content, cast, prev):
            continue
        if cast.at is not None:
            return cast, cast.at
        if cast.figure is not None:
            return cast, figures.state_of(state, content, cast.figure).location
        place = part.location
        if place == NEAREST_BASE:
            place = summons_place(state, content, promo.model_copy(update={"location": NEAREST_BASE}))
        return cast, place
    return None


def _leg_text(content: Content, cast: PromotionCast, location: str) -> str:
    return cast.summons_text.replace("{據點}", content.locations[location].name)


def _refresh(state: GameState, content: Content, promo: PromotionDef) -> list[str]:
    """手上的多段召見照當下重挑一次版本：換了版本或地點就改寫召見、回傳新的那一句（召見自動改由接手的人發）；
    沒有成立的版本時照舊等著。"""
    s = state.player.summons
    found = current_cast(state, content, promo, s.leg, s.prev)
    if found is None:
        return []
    cast, location = found
    if (cast.event, location) == (s.event, s.location):
        return []
    s.event, s.location, s.figure = cast.event, location, cast.figure
    return [_leg_text(content, cast, location)]

def check_summons(state: GameState, content: Content) -> list[str]:
    """該不該發召見（每次行動、同步的最後檢查）：第一季、有陣營、這一季還沒落幕、還沒拿到第 4 階資格。
    手上有多段召見的，照當下重挑版本（_refresh）。沒有召見時看下一階：本季貢獻到 threshold；第 3、4 階還要完成過那一階
    的一種機緣——貢獻到了、機緣還沒有，那一階說一次 HINT。第 3、4 階挑得到版本才發（沒有人能出面就先不發）。"""
    p, w = state.player, state.world
    if not season_one(content, w) or p.faction is None or w.ended or p.qualified:
        return []
    if p.summons is not None:
        promo = _pending(state, content)
        return _refresh(state, content, promo) if promo is not None and promo.legs else []
    rank = rank_of(state) + 1
    promo = promotion_for(content, p.faction, rank)

def next_leg(state: GameState, content: Content, from_event: str) -> list[str]:
    """這一段演完（選項效果的 summons_next）：召見往下一段，挑版本、改寫地點，回傳下一段的召見那一句。
    下一段此刻沒有成立的版本時，召見留在這一段之後等著（地點空著，之後 check_summons 會補上）。"""
    s = state.player.summons
    promo = _pending(state, content)
    if s is None or promo is None or not promo.legs:
        return []
    s.leg, s.prev, s.event, s.location = s.leg + 1, from_event, None, ""
    found = current_cast(state, content, promo, s.leg, s.prev)
    if found is None:
        return []
    cast, location = found
    s.event, s.location, s.figure = cast.event, location, cast.figure
    return [_leg_text(content, cast, location)]

def _bot_opportunity(state: GameState, content: Content, rank: int) -> bool:
    """伺服器假人不做機緣（bot_policy 不選 opp:），貢獻到了、那一階還沒有機緣時，每個季曆日擲一次 bot_opp_chance，
    中了就記成完成一種那一階的機緣（企劃者 2026-10-06）。骰子用假人自己的種子＋季＋季曆日＋階，所以同一天多叫幾次
    也只擲一次、結果固定。不發任何訊息：之後的召見、應召、奇遇、軍情都照真人的路走。記到了回 True。"""
    p = state.player
    if p.bot is None:
        return False
    day = calendar.point(state.world.time, content, state.world).cal_day
    if p.bot_opp_day == day:
        return False
    p.bot_opp_day = day
    roll = random.Random(f"{p.bot.seed}:{p.season_number}:{day}:{rank}")
    if roll.random() >= content.config.bot_opp_chance:
        return False
    left = [o.id for o in content.opportunities if o.faction == p.faction and o.rank == rank and o.id not in p.opp_done]
    if not left:
        return False
    p.opp_done.append(roll.choice(left))
    return True
```

def plot_win_here(state: GameState, content: Content, loc_id: str) -> bool:
    """在這裡遊歷打贏，會不會替自己參加的 win 類密謀記一處：那條戰線那一路還沒人辦、自己還沒做滿上限、這裡打得到敵方隊伍。"""
    from .orders import siege_places  # noqa: PLC0415

    p = state.player
    front = front_of(content, loc_id)
    for plot in _my_plots(state):
        o = _plot_def(content, plot)
        if o.plot.how != "win" or _done_by(plot, p.name) >= _cap_per_person(content, o):
            continue
        if any(x.front == front and x.key not in plot.parts for x in o.plot.parts) \
                and loc_id in siege_places(content, p.faction, front):
            return True
    return False


def plot_targets(state: GameState, content: Content) -> list[str]:
    """還要去辦密謀的地點：check 類是還沒人辦的那幾處；win 類是還缺的那幾條戰線上打得到敵方隊伍的地點。
    自己已經做滿上限的那場不算。照內容檔的地點順序。"""
    from .orders import siege_places  # noqa: PLC0415

    p = state.player
    wanted: set[str] = set()
    for plot in _my_plots(state):
        o = _plot_def(content, plot)
        if _done_by(plot, p.name) >= _cap_per_person(content, o):
            continue
        for part in o.plot.parts:
            if part.key in plot.parts:
                continue
            if part.at is not None:
                wanted.add(part.at)
            elif part.front is not None:
                wanted |= set(siege_places(content, p.faction, part.front))
    return [loc for loc in content.locations if loc in wanted]
```

def _bot_roll(game: Game, profile: BotProfile, key: str, chance: float) -> bool:
    """假人自己的骰子（同 ranks._bot_opportunity）：種子＋這個角色的季＋key，同一個 key 永遠同一個結果。"""
    roll = random.Random(f"{profile.seed}:{game.state.player.season_number}:{key}")
    return roll.random() < chance


def _plot_score(game: Game, arg: str, profile: BotProfile) -> float | None:
    """集體密謀的三個選項（正式版丙一 Task 5）：請命每個季曆日擲一次 bot_plot_chance、響應每一場擲一次 bot_join_chance，
    中了才給 PLOT_SCORE；人在那一處就去辦。其他機緣（送東西、試、拼圖、指認……）回 None，照舊不做。"""
    what, _, rest = arg.partition(":")
    s, c = game.state, game.content
    if what == "plot":
        day = calendar.point(s.world.time, c, s.world).cal_day
        return PLOT_SCORE if _bot_roll(game, profile, f"plot:{rest}:{day}", c.config.bot_plot_chance) else None
    if what == "join":
        return PLOT_SCORE if _bot_roll(game, profile, f"join:{rest}", c.config.bot_join_chance) else None
    if what == "part":
        return PLOT_SCORE
    return None
```
