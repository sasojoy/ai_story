def _game(content, name="甲", faction=None, ally=None, at=None):
    game = Game.new(content, name, rng=random.Random(0))
    p = game.state.player
    p.faction, p.ally = faction, ally
    if at is not None:
        p.location = at
    return game

    assert real_content().config.haoqiang_allied is False

    assert game.state.world.haoqiang_allied is True
    assert allied.on(wk, game.state.world) and rules.haoqiang_allied(wk, game.state.world)


def test_fourteen_day_season_keeps_the_third_party():
    c = real_content()
    c.config.auto_open_first_season = True
    c.config.haoqiang_allied = True  # 設定打開了，可是 14 天的季沒有第一季的章
    game = _game(c, faction="haoqiang", ally="guan")
    assert not allied.on(c, game.state.world)
    assert allied.side_of(game.state, c) == "haoqiang"


def test_side_of(wk):
    game = _game(wk, faction="haoqiang", ally="huang")
    assert allied.side_of(game.state, wk) == "huang"
    game.state.player.faction, game.state.player.ally = "guan", None
    assert allied.side_of(game.state, wk) == "guan"
    game.state.player.faction = None
    assert allied.side_of(game.state, wk) is None


def test_needs_ally(wk):
    game = _game(wk, faction="haoqiang")
    assert allied.needs_ally(game.state, wk)
    game.state.player.ally = "guan"
    assert not allied.needs_ally(game.state, wk)
    off = third_party_gentry(real_content("weekend"))
    off.config.auto_open_first_season = True
    assert not allied.needs_ally(_game(off, name="乙", faction="haoqiang").state, off)


def test_roll_keys():
    assert allied.roll_key("guan", None) == "guan"
    assert allied.roll_key("haoqiang", "huang") == "haoqiang@huang"
    assert allied.split_key("haoqiang@huang") == ("haoqiang", "huang")
    assert allied.side_of_key("haoqiang@huang") == "huang" and allied.side_of_key("guan") == "guan"
    assert allied.side_of_key(None) is None
    rolls = {"guan": 9, "haoqiang@guan": 3, "huang": 7, "haoqiang@huang": 2, "haoqiang": 1}
    assert allied.faction_totals(rolls) == {"guan": 9, "haoqiang": 6, "huang": 7}

def allies_of(content: Content, faction_id: str | None) -> list[FactionDef]:
    """這個陣營可以依附的幾軍（照劇本的順序）；不是要依附的陣營是空的。"""

def side_of(state: GameState, content: Content) -> str | None:
    """敵我的一個讀法：依附的季、依附了一軍的人是那一軍；其他人是自己的陣營（散人 None）。"""
    p = state.player

    p = state.player

def roll_key(faction: str, ally: str | None) -> str:
    """名冊的鍵：依附了一軍是「陣營@那一軍」，其他是陣營 id。"""
    return f"{faction}{SEP}{ally}" if ally else faction


def split_key(key: str) -> tuple[str, str | None]:
    faction, _, ally = key.partition(SEP)
    return faction, ally or None


def side_of_key(key: str | None) -> str | None:
    """名冊的鍵 → 算哪一方。"""
    if key is None:
        return None
    faction, ally = split_key(key)
    return ally or faction


def faction_totals(rolls: dict[str, int]) -> dict[str, int]:
    """名冊照陣營（身分）數：補假人、腳本照它。"""
    out: dict[str, int] = {}
    for key, n in rolls.items():
        faction = split_key(key)[0]
        out[faction] = out.get(faction, 0) + n
    return out


def side_totals(rolls: dict[str, int]) -> dict[str, int]:
    """名冊照那一方數：名冊那一句、假人選邊照它。"""
    out: dict[str, int] = {}
    for key, n in rolls.items():
        side = side_of_key(key) or key
        out[side] = out.get(side, 0) + n
    return out


def gentry_totals(rolls: dict[str, int]) -> dict[str, int]:
    """每一軍裡有幾個依附的人（名冊那一句的括號）。"""
    out: dict[str, int] = {}
    for key, n in rolls.items():
        ally = split_key(key)[1]
        if ally is not None:
            out[ally] = out.get(ally, 0) + n
    return out


def hidden_trends(content: Content, world: WorldState) -> set[str]:
    """依附的季不存在的大勢線：只有要依附的陣營（豪強）當目標的線（割據）。"""
    if not on(content, world):
        return set()
    armies = [f for f in content.scenario.factions if not f.allies]
    gentry = [f for f in content.scenario.factions if f.allies]
    return {t for f in gentry for t in f.goals if not any(t in a.goals for a in armies)}

ALLY_PREFIX = "faction:ally:"


def pick_side(name: str, option_ids: list[str], rolls: dict[str, int]) -> str | None:
    """假人選邊（設計第十三節）：選單上的選邊鈕裡，那一軍此刻人少的（照 side_totals，跟名冊那一句同一個數）；一樣多照名號雜湊，
    不碰任何亂數。沒有選邊鈕是 None。"""
    picks = [i for i in option_ids if i.startswith(ALLY_PREFIX)]
    if not picks:
        return None
    sides = side_totals(rolls)

    def key(option_id: str) -> tuple[int, str]:
        side = option_id.removeprefix(ALLY_PREFIX)
        return sides.get(side, 0), hashlib.sha256(f"{name}|{side}".encode()).hexdigest()

    return min(picks, key=key)
