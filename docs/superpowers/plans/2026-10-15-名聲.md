"""名聲（核心玩法重整第 3 塊）的共用地基：段、稱號、季長倍數、開關。"""
import random

from conftest import real_content, renown_on

from tianxia import renown
from tianxia.state import new_game_state
from tianxia.world_state import fresh_season, length_factor, stamp_season


def _state(content, fame=0, good=0, evil=0):
    s = new_game_state(content, "甲")
    s.player.stats.update({"fame": fame, "good": good, "evil": evil})
    return s


def test_fame_has_five_tiers_and_titles_show_from_the_first():
    c = real_content("weekend")
    assert [renown.fame_tier(_state(c, fame=f), c) for f in (0, 9, 10, 24, 25, 49, 50, 89, 90)] == [0, 0, 1, 1, 2, 2, 3, 3, 4]
    assert renown.fame_title(_state(c, fame=27), c) == "名動一方"
    assert renown.fame_title(_state(c, fame=3), c) is None


def test_virtue_follows_the_higher_book_and_ties_go_to_xia():
    c = real_content("weekend")
    assert renown.path(_state(c, good=18, evil=3)) == "俠"
    assert renown.path(_state(c, good=4, evil=22)) == "寇"
    assert renown.path(_state(c, good=6, evil=6)) == "俠"
    assert renown.virtue_title(_state(c, good=18, evil=3), c) == "一方豪俠"
    assert renown.virtue_title(_state(c, good=4, evil=22), c) == "江湖巨寇"
    assert renown.virtue_title(_state(c, good=2, evil=1), c) is None


def test_women_get_the_female_title():
    c = real_content("weekend")
    s = _state(c, good=0, evil=6)
    s.player.gender = "女"
    assert renown.virtue_title(s, c) == "綠林女傑"


def test_titles_lists_fame_then_virtue_from_a_given_fame_tier():
    c = real_content("weekend")
    s = _state(c, fame=12, good=4, evil=22)
    assert renown.titles(s, c) == ["小有名氣", "江湖巨寇"]
    assert renown.titles(s, c, fame_from=2) == ["江湖巨寇"]


def test_the_scale_multiplies_thresholds_from_the_stamp():
    c = real_content()  # 14 天
    season = fresh_season(c)
    assert season.renown_scale == length_factor(season, c) == 5.6
    s = _state(c, fame=27)
    s.world = season
    assert renown.fame_bars(c, season) == [56, 140, 280, 504]
    assert renown.fame_tier(s, c) == 0
    assert renown.virtue_bars(c, season) == [28, 84, 168, 280]


def test_a_season_without_the_stamp_counts_as_one():
    c = real_content()
    season = fresh_season(c)
    season.renown_scale = None  # 上線那一刻正在跑的季
    assert renown.scale(season) == 1.0 and renown.fame_bars(c, season) == [10, 25, 50, 90]


def test_switches_off_mean_no_titles_and_no_stamp():
    c = real_content("weekend")
    c.config.renown.titles = False
    s = _state(c, fame=99, good=60)
    assert renown.fame_tier(s, c) == renown.virtue_tier(s, c) == 0 and renown.titles(s, c) == []
    c.config.renown.titles = True
    c.config.renown.scale = False
    season = fresh_season(c)
    stamp_season(season, c)
    assert season.renown_scale is None


def test_fixture_content_has_every_switch_off_and_renown_on_turns_them_on(content):
    cfg = content.config.renown
    assert not cfg.titles and not cfg.official_wanted
    renown_on(content)
    assert cfg.titles and cfg.official_wanted and renown.on(content, "duel")


def test_discount_words():
    assert [renown.discount_word(f) for f in (0.97, 0.94, 0.9, 0.85, 1.2)] == ["九七折", "九四折", "九折", "八五折", "貴兩成"]


def test_price_factor_follows_the_fame_tier():
    c = real_content("weekend")
    assert [renown.price_factor(_state(c, fame=f), c) for f in (0, 10, 25, 50, 90)] == [1.0, 0.97, 0.94, 0.9, 0.85]
    c.config.renown.discount = False
    assert renown.price_factor(_state(c, fame=90), c) == 1.0

def on(content: Content, switch: str = "titles") -> bool:
    """這一樣開著沒：titles 關著時一律沒有。"""
    cfg = content.config.renown
    return cfg.titles and (switch == "titles" or bool(getattr(cfg, switch)))


def scale(world: WorldState) -> float:
    """門檻乘幾倍：開季時蓋的章（world_state.stamp_season）；沒有章（上線那一刻正在跑的季、scale 關著）是 1。"""
    return world.renown_scale or 1.0


def _bars(base: list[int], world: WorldState) -> list[int]:
    k = scale(world)
    return [max(1, int(b * k + 0.5)) for b in base]  # 四捨五入照專案的寫法（round 逢 .5 取偶數）


def fame_bars(content: Content, world: WorldState) -> list[int]:
    return _bars(content.config.renown.fame_thresholds, world)


def virtue_bars(content: Content, world: WorldState) -> list[int]:
    return _bars(content.config.ranger.thresholds, world)


def tier_of(value: int, bars: list[int]) -> int:
    return sum(1 for need in bars if value >= need)


def fame_tier(state: GameState, content: Content) -> int:
    if not on(content):
        return 0
    return tier_of(state.player.stats.get("fame", 0), fame_bars(content, state.world))

    return "寇" if st.get("evil", 0) > st.get("good", 0) else "俠"

        return 0
    return tier_of(virtue_value(state), virtue_bars(content, state.world))

    """稱號的短版：名望（fame_from 段起才寫）在前、善惡（第 1 段起）在後；沒有的不寫。"""
    out = []

        out.append(FAME_TITLES[fame_tier(state, content)])
    virtue = virtue_title(state, content)
    if virtue:
        out.append(virtue)
    return out


def meets(state: GameState, content: Content, way: str, tier: int) -> bool:
    """走 way 那一條路、到第 tier 段了沒（俠寇選項、正邪武學、官府通緝共用）。"""
    return path(state) == way and virtue_tier(state, content) >= tier


def discount_word(factor: float) -> str:
    """0.94 → 「九四折」、0.9 → 「九折」、1.2 → 「貴兩成」。"""
    pct = int(factor * 100 + 0.5)
    if pct < 100:
        tens, ones = divmod(pct, 10)
        return f"{_CN[tens]}{_CN[ones] if ones else ''}折"
    extra = pct - 100
    return f"貴{'兩' if extra == 20 else _CN[extra // 10]}成" if extra % 10 == 0 else f"貴{extra}%"


def price_factor(state: GameState, content: Content) -> float:
    """名望那一半的店家折扣（4.1）：第 1～4 段各一個倍數；沒到、開關關著是 1。地盤那一半另外乘（territory.price_here）。"""
    t = fame_tier(state, content) if on(content, "discount") else 0
    return content.config.renown.discounts[t - 1] if t > 0 else 1.0

def gap_note(state: GameState, content: Content, key: str) -> str:
    """名望、善名、惡名加了之後接的那一句（設計 6.4）。"""
    names = content.config.stat_names

        t, bars, n = fame_tier(state, content), fame_bars(content, state.world), state.player.stats.get("fame", 0)
        return f"（已是{FAME_TITLES[4]}）" if t >= 4 else f"（離「{FAME_TITLES[t + 1]}」還差 {bars[t] - n}）"
    way = path(state)

def annotate(changes: list[str], state: GameState, content: Content) -> list[str]:
    """同一個標籤（名望、善名、惡名）的數字合成一行，排在第一次出現的地方；總和是正的接差距，扣的不接，總和 0 的拿掉。
    其他行照原樣。已經接過差距的行（有括號）不動：同一下呼叫兩次也一樣。"""
    if not on(content):
        return list(changes)
    keys = {content.config.stat_names.get(k, k): k for k in ("fame", "good", "evil")}
    out: list[str | None] = []
    totals: dict[str, int] = {}
    slot: dict[str, int] = {}
    for line in changes:
        m = _PLAIN.match(line)
        key = keys.get(m.group(1)) if m else None
        if key is None:
            out.append(line)
            continue
        if key not in slot:
            slot[key] = len(out)
            out.append(None)
            totals[key] = 0
        totals[key] += int(m.group(3)) * (-1 if m.group(2) == "-" else 1)
    for key, at in slot.items():
        total = totals[key]
        if total:
            label = content.config.stat_names.get(key, key)
            out[at] = f"{label} {total:+d}" + (gap_note(state, content, key) if total > 0 else "")
    return [x for x in out if x is not None]

    day 是此刻的遊戲日（rules.game_day）。上線之後第一次看（renown_seen 是 None）只記段。開關關著什麼都不做。"""
    out = Crossing()
    if not on(content):
        return out
    p = state.player
    f, way, v = fame_tier(state, content), path(state), virtue_tier(state, content)
    seen = p.renown_seen
    if seen is None:
        p.renown_seen = RenownSeen(fame=f, path=way, virtue=v, best_fame=f,
                                   best_xia=v if way == "俠" else 0, best_kou=v if way == "寇" else 0)
        out.first = True
        return out
    if f > seen.best_fame:
        out.lines += [UP_FAME.format(title=FAME_TITLES[f]), perk_line(fame_perks(content, f))]
        out.up.append(FAME_TITLES[f])
        out.annals.append((UP_FAME.format(title=FAME_TITLES[f]), 3 if f >= 3 else 2))
        seen.best_fame = f
    elif f < seen.fame and seen.fame_drop_day != day:
        out.lines.append(DOWN.format(title=FAME_TITLES[seen.fame]))
        seen.fame_drop_day = day
    best = seen.best_xia if way == "俠" else seen.best_kou
    if way != seen.path and seen.virtue > 0 and v > 0 and seen.path_day != day:
        old = named(VIRTUE_TITLES[seen.path][seen.virtue], state)
        out.lines.append(SWITCH.format(old=old, new=named(VIRTUE_TITLES[way][v], state)))
        seen.path_day = day
    if v > best:
        title = named(VIRTUE_TITLES[way][v], state)
        if way == seen.path or seen.virtue == 0:
            out.lines.append(UP_VIRTUE.format(title=title))
        out.lines.append(perk_line(virtue_perks(content, way, v)))
        out.up.append(title)
        out.annals.append((UP_VIRTUE.format(title=title), 3 if v >= 3 else 2))
        if way == "俠":
            seen.best_xia = v
        else:
            seen.best_kou = v
    seen.fame, seen.path, seen.virtue = f, way, v
    out.lines = [line for line in out.lines if line]
    return out
```

"""陣營通緝（名聲 4.6）與官府通緝（5.4）。"""
import random

from conftest import real_content

from tianxia import bounties
from tianxia.engine import Game


def _game(name, faction=None, fame=0, good=0, evil=0):
    c = real_content("weekend")
    c.config.auto_open_first_season = True
    g = Game.new(c, name, rng=random.Random(0))
    g.client = None
    p = g.state.player
    p.faction = faction
    p.stats.update({"fame": fame, "good": good, "evil": evil})
    p.joined_at = -1e9  # 過了新手期
    return g


def test_faction_wanted_posts_once_a_week_on_ones_own_turn():
    g = _game("甲", faction="guan", fame=50)
    g.own_turn = True
    assert bounties.post_auto(g.state, g.content) == ["黃巾軍貼出了通緝你的告示（威震州郡）。"]
    assert bounties.post_auto(g.state, g.content) == []
    posted = [b for b in g.state.world.bounties if b.auto_by == "陣營"]
    assert len(posted) == 1 and posted[0].faction == "huang" and posted[0].silver == 40


def test_official_wanted_for_great_rogues_any_side():
    g = _game("乙", evil=15)
    assert bounties.post_auto(g.state, g.content) == ["官府貼出了海捕文書，畫的是你（江湖巨寇）。"]
    b = next(x for x in g.state.world.bounties if x.auto_by == "官府")
    hero = _game("丙", faction="huang", good=15)
    hero.state.world = g.state.world
    assert bounties.take_problem(hero.state, hero.content, b) is None
    assert bounties.issuer(hero.content, b) == "官府"
    assert bounties.stranded(hero.state, hero.content, "guan") == []  # 換陣營不作廢


def test_system_wanted_expire_off_the_board():
    g = _game("甲", faction="guan", fame=50)
    bounties.post_auto(g.state, g.content)
    b = next(x for x in g.state.world.bounties if x.auto_by)
    g.state.world.time = b.expires + 1
    g.state.world.bounty_week = 0
    week = bounties.calendar.point(g.state.world.time, g.content, g.state.world).week
    bounties.issue(g.state, g.content, week)
    assert b not in g.state.world.bounties
```
