```python
見 docs/superpowers/specs/2026-10-03-地圖擴充與移動-design.md 第四、五節。"""
from __future__ import annotations

import heapq
import math
from pathlib import Path

import pytest

from tianxia.atlas import region_of
from tianxia.content import load_content

CONTENT_DIR = Path(__file__).parent.parent / "content"

REGIONS = {
    "youzhou": {"zhuo_county", "market_town", "loushang_village", "zhuo_militia_hall", "juma_river", "yanshan_foot"},

@pytest.fixture(scope="module")
def content():
    return load_content(CONTENT_DIR)


def _region(loc_id: str) -> str:
    return next(r for r, members in REGIONS.items() if loc_id in members)


def _roads(content) -> dict[frozenset, str]:
    """每條路一次：{frozenset({a, b}): 路的種類}。connections 的每一項是字串（一般路）或帶 to／road 的物件。"""
    roads = {}
    for loc in content.locations.values():
        for c in loc.connections:
            to, road = (c, "路") if isinstance(c, str) else (c.to, c.road)
            roads[frozenset({loc.id, to})] = road
    return roads


def _minutes(content, pair: frozenset, road: str) -> float:
    a, b = (content.locations[i] for i in pair)
    return math.dist((a.x, a.y), (b.x, b.y)) * content.config.road_factor[road] * content.config.travel_minutes_per_unit


def _connected(nodes: set[str], roads) -> bool:
    start = next(iter(nodes))
    seen, stack = {start}, [start]
    while stack:
        here = stack.pop()
        for pair in roads:
            if here in pair:
                (other,) = pair - {here}
                if other in nodes and other not in seen:
                    seen.add(other)
                    stack.append(other)
    return seen == nodes


def _shortest(content, roads, src: str, dst: str) -> float:
    adj: dict[str, list[tuple[str, float]]] = {}
    for pair, road in roads.items():
        a, b = tuple(pair)
        m = _minutes(content, pair, road)
        adj.setdefault(a, []).append((b, m))
        adj.setdefault(b, []).append((a, m))
    best, heap = {src: 0.0}, [(0.0, src)]
    while heap:
        d, here = heapq.heappop(heap)
        if here == dst:
            return d
        for other, m in adj[here]:
            if d + m < best.get(other, math.inf):
                best[other] = d + m
                heapq.heappush(heap, (d + m, other))
    return math.inf


def test_the_map_has_forty_locations_in_five_regions(content):
    assert set(content.locations) == set().union(*REGIONS.values())
    assert NEW_LOCATIONS <= set(content.locations)
    assert {r.id for r in content.map.regions} == set(REGIONS)
    for loc_id in content.locations:
        assert region_of(content, loc_id).id == _region(loc_id), loc_id


def test_every_region_is_connected_and_has_a_loop(content):
    roads = _roads(content)
    for region, members in REGIONS.items():
        inside = {pair: road for pair, road in roads.items() if pair <= members}
        assert _connected(members, inside), region
        assert len(inside) >= len(members), f"{region} 沒有環"  # 連通的圖，邊數 ≥ 點數才有環


def test_cross_region_roads_are_exactly_the_planned_ones(content):
    roads = _roads(content)
    cross = {pair: road for pair, road in roads.items() if len({_region(i) for i in pair}) == 2}
    assert {pair: road for pair, road in cross.items()} == {pair: road for pair, (road, _) in CROSS_ROADS.items()}


def test_cutting_any_cross_region_road_keeps_the_map_connected(content):
    roads = _roads(content)
    for pair in CROSS_ROADS:
        rest = {p: r for p, r in roads.items() if p != pair}
        assert _connected(set(content.locations), rest), sorted(pair)


def test_cross_region_roads_take_the_planned_minutes(content):
    for pair, (road, minutes) in CROSS_ROADS.items():
        assert minutes * 0.8 <= _minutes(content, pair, road) <= minutes * 1.2, sorted(pair)


def test_hops_inside_a_region_take_two_to_four_minutes(content):
    for pair, road in _roads(content).items():
        if pair in CROSS_ROADS:
            continue
        assert 1.6 <= _minutes(content, pair, road) <= 4.8, sorted(pair)


def test_the_longest_trips(content):
    roads = _roads(content)
    assert 24 <= _shortest(content, roads, "guangzong", "wan_city") <= 42  # 設計：約 30～35 分鐘，±20%
    assert 36 <= _shortest(content, roads, "zhuo_county", "wan_city") <= 54  # 設計：約 45 分鐘，±20%


def test_locations_keep_room_for_their_labels(content):
    locs = list(content.locations.values())
    for i, a in enumerate(locs):
        for b in locs[i + 1:]:
            assert math.dist((a.x, a.y), (b.x, b.y)) >= 60, (a.id, b.id)


def test_new_locations_have_a_description_and_tags(content):
    for loc_id in NEW_LOCATIONS:
        loc = content.locations[loc_id]
        assert len(loc.description) >= 20 and loc.tags, loc_id
```

**Files（每個任務）：** Create 該任務的事件檔與測試檔，**不動其他任何檔案**（四個任務平行做也不會互撞）。

**Interfaces:**
- Consumes：Task 1 的地點 id、Task 2 的隊伍 id（`combat` 只能用 `content/squads.json` 裡有的 id）。
- Produces：事件 id 一律以大區英文名加底線開頭（`youzhou_`、`jizhou_`、`luoyang_`、`nanyang_`）。

**事件的寫法（照現有 `content/events/kou.json` 的格式）：**
- 每個新地點至少 1 則、最多 2 則探索事件（`actions` 不寫＝預設 `["explore"]`；`locations` 只放那一個地點）。
- `title` 2～6 個字；`text` 1～3 句；每則 2～3 個選項。
- 至少一半的事件要有一個選項帶 `check`（`stat` 用 `str`／`agi`／`con`／`wis`，`difficulty` 5～8，照現有事件的範圍）或 `combat`（隊伍 id）；有 `check` 或 `combat` 的選項要寫 `fail_effect`。
- `effect.stats` 只用 `silver`、`good`、`evil`、`fame`、`str`、`agi`、`con`、`wis`；數量：`silver` ≤ 40、`fame` −3～+3、`good`／`evil` ≤ 3、四圍 ≤ 1。`trend` 只用 `huangjin`，±1～3。`materials` 可以給一階或二階素材（`gang_1`、`gang_2`、`rou_1`、`rou_2`、`kuai_1`、`kuai_2`、`man_1`、`man_2`），每則最多 1 個。
- 不新增任何旗標（`flags_add`、`world_flags_add`），不寫 `next_event`、`recruit`、`join_sect`。
- `rumor` 每個檔最多 2 則，用 `{name}` 代名號，不得透露伏筆。
- 內容貼著那個地點的描寫與 184 年的處境（例：鉅鹿道壇是受符水入道、廣宗是張角坐鎮的城、盧植營是投軍、孟津渡有西涼兵馬、北邙山有盜墓的人、新野是逃難的百姓）。口吻是漢末戰亂，不是武俠單挑。人物、武學不用金庸等作品的名字。
- 範例（格式示意，不要照抄內容）：

```json
{
  "id": "jizhou_fushui",
  "title": "壇前符水",
  "locations": ["julu_altar"],
  "text": "道壇前排著長長的人龍，黃衣道士逐一遞上符水。有個老婦人捧著碗遲遲不肯喝，回頭望著你，像是想問什麼。",
  "choices": [
    {
      "text": "勸她喝下，說這是求平安",
      "effect": {"text": "老婦人喝了符水，向你道謝，口中念著「黃天當立」。", "stats": {"evil": 1}, "trend": {"huangjin": 1}}
    },
    {
      "text": "低聲問她家中出了什麼事",
      "check": {"stat": "wis", "difficulty": 6},
      "effect": {"text": "她說兒子被拉進了營裡。你記下了名字，答應替她打聽。", "stats": {"good": 2, "fame": 1}},
```

- [ ] **Step 1: 寫失敗的測試** — 建立該任務的測試檔（以冀州為例；其他任務把 `REGION`、`PREFIX`、`LOCATIONS`、`MIN_EVENTS`、`MAX_EVENTS` 換成表上的值）：

```python
"""冀州新地點的探索事件（地圖擴充設計第七節；計畫 Task 6）。"""
from __future__ import annotations

import json
from pathlib import Path

import pytest

from tianxia.content import load_content

CONTENT_DIR = Path(__file__).parent.parent / "content"
EVENT_FILE = CONTENT_DIR / "events" / "jizhou.json"
PREFIX = "jizhou_"
LOCATIONS = {"julu_altar", "guangzong", "luzhi_camp", "xiaquyang", "haozu_fort", "baima_ford"}
MIN_EVENTS, MAX_EVENTS = 7, 9
STAT_KEYS = {"silver", "good", "evil", "fame", "str", "agi", "con", "wis"}


@pytest.fixture(scope="module")
def content():
    return load_content(CONTENT_DIR)


@pytest.fixture(scope="module")
def events():
    return json.loads(EVENT_FILE.read_text(encoding="utf-8"))


def test_every_new_location_has_one_or_two_explore_events(events):
    assert MIN_EVENTS <= len(events) <= MAX_EVENTS
    per_place = {loc: 0 for loc in LOCATIONS}
    for e in events:
        assert e["id"].startswith(PREFIX), e["id"]
        assert e.get("actions", ["explore"]) == ["explore"], e["id"]
        (loc,) = e["locations"]
        per_place[loc] += 1
    assert all(1 <= n <= 2 for n in per_place.values()), per_place


def test_events_load_and_only_fight_known_squads(content, events):
    for e in events:
        assert e["id"] in content.events
        for choice in e["choices"]:
            if "combat" in choice:
                assert choice["combat"] in content.squads, (e["id"], choice["combat"])


def test_events_stay_within_the_reward_limits(events):
    risky = 0
    rumors = 0
    for e in events:
        assert 2 <= len(e["choices"]) <= 3, e["id"]
        if any("check" in c or "combat" in c for c in e["choices"]):
            risky += 1
        for c in e["choices"]:
            if "check" in c or "combat" in c:
                assert "fail_effect" in c, e["id"]
            if "check" in c:
                assert c["check"]["stat"] in {"str", "agi", "con", "wis"} and 5 <= c["check"]["difficulty"] <= 8, e["id"]
            for key in ("effect", "fail_effect"):
                eff = c.get(key, {})
                stats = eff.get("stats", {})
                assert set(stats) <= STAT_KEYS, e["id"]
                assert stats.get("silver", 0) <= 40 and -3 <= stats.get("fame", 0) <= 3, e["id"]
                assert stats.get("good", 0) <= 3 and stats.get("evil", 0) <= 3, e["id"]
                assert all(stats.get(k, 0) <= 1 for k in ("str", "agi", "con", "wis")), e["id"]
                assert set(eff.get("trend", {})) <= {"huangjin"} and all(1 <= abs(v) <= 3 for v in eff.get("trend", {}).values()), e["id"]
                assert len(eff.get("materials", {})) <= 1, e["id"]
                assert not eff.get("flags_add") and not eff.get("world_flags_add"), e["id"]
                assert not eff.get("next_event") and not eff.get("recruit") and not eff.get("join_sect"), e["id"]
                rumors += bool(eff.get("rumor"))
    assert risky * 2 >= len(events)
    assert rumors <= 2
```
