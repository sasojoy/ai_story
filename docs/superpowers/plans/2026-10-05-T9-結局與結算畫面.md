- [ ] **Step 5：結局判定**（`tianxia/world.py`，取代 `evaluate_ending`）

```python
def season_endings(state: GameState, content: Content) -> list[Ending]:
    """這一季算的結局：第一季（開關開著＋這一季的章）只看標了 season_one 的，beta 季只看沒標的。清單照劇本的順序。"""
    on = rules.season_one(content, state.world)
    return [e for e in content.scenario.endings if e.season_one == on]


def stance_holds(ending: Ending, stances: dict[str, int]) -> bool:
    """結局的態勢條件（第一季設計 4.4、13.1）：至少、至多，以及「這一方最高」（平手也算最高，順序交給清單）。"""
    if any(stances[side] < value for side, value in ending.stance_min.items()):
        return False
    if any(stances[side] > value for side, value in ending.stance_max.items()):
        return False
    return ending.stance_top is None or stances[ending.stance_top] >= max(stances.values())


def evaluate_ending(state: GameState, content: Content) -> Ending:
    """照清單順序第一個成立的結局；都不成立就是這一季清單的最後一筆（保底）。第一季另外看三方態勢（rules.stances）。"""
    endings = season_endings(state, content)
    stances = rules.stances(state, content) if rules.season_one(content, state.world) else {}
    for ending in endings:
        if ending.storyline not in (None, state.world.storyline):
            continue
        if check_condition(ending.condition, state) and (not ending.season_one or stance_holds(ending, stances)):
            return ending
    return endings[-1]


def decisive_ending(state: GameState, content: Content) -> Ending | None:
```

```python
def contribution_rankings(
    content: Content, world: WorldStateStore, characters: CharacterStore | None = None,
) -> dict[str, list[tuple[str, int]]]:
    """第一季結算畫面的各陣營出力前五（計畫 T9、第一季設計 13.3）：這一季的角色、有陣營、貢獻大於 0，照貢獻排。

    from .rules import display_name  # noqa: PLC0415

    season_number = world.get_season_number()
    ranks: dict[str, list[tuple[str, int]]] = {f.id: [] for f in content.scenario.factions}
    for state in (characters or open_characters()).all():

        if p.season_number != season_number or p.faction not in ranks or p.contrib <= 0:

        ranks[p.faction].append((display_name(state), p.contrib))

        rows.sort(key=lambda row: row[1], reverse=True)
        ranks[faction] = rows[:TOP_CONTRIB]
    return ranks
```

```python
def season_end_time(season: WorldState, content: Content) -> float:
    """這一季在世界時鐘上何時收：第一季照排定的季末（schedule["finale"]，管理者可改，T10），其他照季長（舊季 14 天）。"""
    if calendar.season_one_on(season, content) and any(e.kind == "finale" for e in content.timetable):
        return season.schedule.get("finale", season_length_days(season, content) * DAY)
    return season_length_days(season, content) * DAY


def _finale(state: GameState, content: Content, ending: Ending, early: bool) -> str:
    """季末公告（時刻表結算第 12 週）：開頭（季中收季是 early_preface）＋結局句＋退場人物的後話（董卓兵敗）。

    w = state.world
    event = next((e for e in content.timetable if e.kind == "finale"), None)
    if event is None or event.id in w.timeline:
        return ""
    head = event.early_preface if early and event.early_preface else event.preface

    text = head + ending.text + after

    add_rumor(state, text, content=content, layer="world")
    if event.ending_chronicle:
        add_chronicle(state, event.ending_chronicle.replace("{結局}", ending.title))
    return text
```
