## Global Constraints

- 每一條整個遊戲只說一次：記在 `PlayerState.hints_seen`，換季保留（設計 5.1、7.1）。
- 序章裡不出提示（序章本身在教）；入伍段、結語、序章的話在框上時，提示排隊等著。

```python
"""碰到才說（新手引導重做設計第五節）：第一次碰到某個玩法時，對話框說一兩句。每一條整個遊戲只說一次。"""
from __future__ import annotations

from collections.abc import Iterable

from .models import Content
from .state import GameState, HintNote


def note_for(state: GameState, content: Content, hint_id: str) -> HintNote | None:
    """這一條此刻由誰說、說什麼；說不了（散人碰到沒有散人版的陣營提示、內容沒有這一條）是 None。"""
    h = content.hints
    hint = next((x for x in h.hints if x.id == hint_id), None)
    if hint is None:
        return None
    if hint.by == "mentor":
        return HintNote(id=hint_id, speaker=h.head, text=hint.text)
    faction = state.player.faction
    if faction is not None and faction in hint.texts:
        e = content.tutorial.enlist
        who = e.recruiters.get(faction) if e is not None else None
        name = who.name if who is not None else next((f.name for f in content.scenario.factions if f.id == faction), faction)
        return HintNote(id=hint_id, speaker=name, text=hint.texts[faction])
    if faction is None and hint.drifter:
        return HintNote(id=hint_id, speaker=h.head, text=hint.drifter)
    return None


def queue(state: GameState, content: Content, hint_ids: Iterable[str]) -> None:
    """把還沒說過的排進去、記成說過。關了提示、假人都不排；說不了的不記（之後還有機會說）。"""
    p = state.player
    if p.hints_off or p.bot:
        return
    for hint_id in hint_ids:
        if hint_id in p.hints_seen:
            continue
        note = note_for(state, content, hint_id)
        if note is not None:
            p.hint_queue.append(note)
            p.hints_seen.add(hint_id)
```

```python
    def _hint_triggers(self) -> list[str]:
        """此刻成立的碰到才說（設計 5.2）：看的都是角色現在的狀態，「第一次」由 hints_seen 管。順序照設計的表。"""
        s, c, p = self.state, self.content, self.state.player
        on: list[str] = []
        if len(p.insights) >= 2:
            on.append("h_merge")
        wugong = team.player_art(s, c, self.world, p.member.wugong_id) if p.member.wugong_id else None
        neigong = team.player_art(s, c, self.world, p.member.neigong_id) if p.member.neigong_id else None
        if wugong and neigong and martial_arts.counters(wugong.attribute, neigong.attribute):
            on.append("h_clash")
        if any(v > 0 for v in p.art_mastery.values()):
            on.append("h_refine_fail")
        if any(r.tier == "落敗" for r in s.battles if r.kind in ("train", "wild", "event")):
            on.append("h_lose")
        if p.member.injury > 0:
            on.append("h_injury")
        if library.lessons_here(s, c):
            on.append("h_basic_art")
        if self._recruit_target() is not None:
            on.append("h_recruit")
        if s.pending_event and c.events[s.pending_event].free_text is not None:
            on.append("h_free_text")
        if p.recent_sights:
            on.append("h_road")
        if library.held_count(s) >= library.cap_of(s, c) - 5:
            on.append("h_cap")
        status = self._battle_status(tick=False)
        watching = status is not None and self._watching_battle(*status)
        if status is not None and p.faction is None and watching:
            on.append("h_spectator")
        if p.fragments:
            on.append("h_foreshadow")
        if p.events_seen:
            on.append("h_event_reveal")
        if p.summons is not None:
            on.append("h_promotion")
        if status is not None and p.faction is not None and status[0].phase == "muster" and not watching:
            on.append("h_showdown")
        if self._figures_here():
            on.append("h_figure")
        return on

    def _check_hints(self) -> None:
        """每次行動、修練頁的動作、同步之後看一遍（新手引導計畫三）。序章裡、假人不看。"""
        p = self.state.player
        if p.bot or p.hints_off or prologue_rules.active(self.state, self.content):
            return
        hints.queue(self.state, self.content, self._hint_triggers())
```
